    USUARIOS {
        bigint_unsigned id PK
        tinyint_unsigned tipo_documento_id FK
        varchar_25 numero_documento
        varchar_80 nombres
        varchar_80 apellidos
        varchar_120 correo UK
        varchar_25 telefono
        varchar_255 password_hash
        tinyint_unsigned rol_id FK
        tinyint_unsigned estado_id FK
        timestamp creado_en
    }

    CUENTAS {
        bigint_unsigned id PK
        bigint_unsigned usuario_id FK
        varchar_20 numero_cuenta UK
        tinyint_unsigned tipo_id FK
        tinyint_unsigned estado_id FK
        varchar_255 pin_hash
        decimal_15_2 saldo
        timestamp creada_en
    }

    OPERACIONES {
        bigint_unsigned id PK
        tinyint_unsigned tipo_id FK
        tinyint_unsigned estado_id FK
        tinyint_unsigned canal_id FK
        bigint_unsigned realizado_por FK
        decimal_15_2 valor
        varchar_255 descripcion
        timestamp creada_en
    }

    MOVIMIENTOS {
        bigint_unsigned id PK
        bigint_unsigned operacion_id FK
        bigint_unsigned cuenta_id FK
        enum sentido
        decimal_15_2 saldo_resultante
        timestamp creada_en
    }

    TRANSFERENCIAS {
        bigint_unsigned operacion_id PK_FK
        bigint_unsigned cuenta_origen_id FK
        bigint_unsigned cuenta_destino_id FK
    }
